import { splitProps, type JSX } from "solid-js";

type BadgeProps = JSX.SvgSVGAttributes<SVGSVGElement> & {
  /** Accessible name. Without it the SVG is treated as decorative. */
  title?: string;
};

export function Badge(props: BadgeProps) {
  const [local, rest] = splitProps(props, ["title"]);
  return (
    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 240 240" width="1em" height="1em" role={local.title ? "img" : undefined} aria-hidden={local.title ? undefined : "true"} {...rest}>
      {local.title ? <title>{local.title}</title> : null}
      <defs>
        <linearGradient id="gradient" x1="0" y1="0" x2="1" y2="1">
          <stop offset="0" stop-color="#6366f1" />
          <stop offset="1" stop-color="#ec4899" />
        </linearGradient>
      </defs>
      <rect x="20" y="20" width="200" height="200" rx="48" fill="url(#gradient)" />
      <path d="M80.12 122.65l28.87 27.46 53.34-57.67" fill="none" stroke="#fff" stroke-width="16" stroke-linecap="round" stroke-linejoin="round" />
      <circle cx="186" cy="54" r="10" fill="#fbbf24" fill-rule="nonzero" />
    </svg>
  );
}
