import type { SVGProps } from "react";

type BadgeProps = SVGProps<SVGSVGElement> & {
  /** Accessible name. Without it the SVG is treated as decorative. */
  title?: string;
};

export function Badge({ title, ...props }: BadgeProps) {
  return (
    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 240 240" width="1em" height="1em" role={title ? "img" : undefined} aria-hidden={title ? undefined : true} {...props}>
      {title ? <title>{title}</title> : null}
      <defs>
        <linearGradient id="gradient" x1="0" y1="0" x2="1" y2="1">
          <stop offset="0" stopColor="#6366f1" />
          <stop offset="1" stopColor="#ec4899" />
        </linearGradient>
      </defs>
      <rect x="20" y="20" width="200" height="200" rx="48" fill="url(#gradient)" />
      <path d="M80.12 122.65l28.87 27.46 53.34-57.67" fill="none" stroke="currentColor" strokeWidth="16" strokeLinecap="round" strokeLinejoin="round" />
      <circle cx="186" cy="54" r="10" fill="currentColor" fillRule="nonzero" />
    </svg>
  );
}
